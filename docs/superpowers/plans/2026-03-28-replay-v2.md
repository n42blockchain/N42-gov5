# Replay-V2: Full Chain Replay with JMT + LtHash Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replay old chain data into a new database with JMT + LtHash state commitment from genesis, Glamsterdam pre-fork, gap-filling empty blocks, and EraE export for BitTorrent distribution.

**Architecture:** A new `replay-v2` CLI command reads blocks from old DB (`d:\mainnet`), filters transactions, builds new headers with JMT root + LtHash root, fills timeline gaps with empty blocks at 8s intervals, and writes everything to a new DB (`d:\mainnetnew`). Post-replay exports EraE segments and creates a snapshot. The new chain activates all forks (including Glamsterdam EIP-7904) from genesis.

**Tech Stack:** Go, MDBX, JMT (Blake3), LtHash (Blake3 XOF), EraE archive format, Zstd compression, protobuf

**Reference:** See `docs/REPLAY_GUIDE.md` for full architecture and configuration details.

## Compression Strategy

对重放产生的数据进行评估，合适的部分使用 Zstd 压缩：

| 数据类型 | 压缩方式 | 原因 |
|---------|---------|------|
| **EraE 段文件** | Zstd Level 3 | 分发体积大（~60-80GB），压缩后 ~20-30GB，BT/HTTP 传输效率提升 2-3x |
| **Snapshot 批量传输** | Zstd Level 3 | 已有实现 (`internal/snapshot/compress.go`)，网络同步用 |
| **JMT 节点 (JMTNode 表)** | **不压缩** | 随机读写热路径，解压延迟不可接受；节点平均 ~40 bytes 压缩收益低 |
| **PlainState 表** | **不压缩** | EVM 执行热路径，MDBX 原生 B+tree 已有页级紧凑 |
| **JMTVersionRoots 索引** | **不压缩** | 固定 40 bytes/条目，cursor seek 需要原始 key 排序 |
| **Block Headers/Bodies** | **不压缩** | MDBX 内存映射直读，压缩增加 CPU 开销 |
| **P2P 网络消息** | Snappy | 已有实现，低延迟优先 |

**压缩工具**：复用 `internal/snapshot/compress.go` 中已有的 Zstd 池化 encoder/decoder。

```go
			}
		})
	}
```

```go
package replay

// CalcGapBlockTimes returns the timestamps for empty blocks that should be
// inserted between prevTime and currTime to maintain continuous block production.
// Returns nil if the gap is within tolerance.
func CalcGapBlockTimes(prevTime, currTime, period, tolerance uint64) []uint64 {
	if prevTime == 0 || currTime <= prevTime {
		return nil
	}
	delta := currTime - prevTime
	if delta <= tolerance {
		return nil
	}
	var times []uint64
	for t := prevTime + period; t < currTime; t += period {
		times = append(times, t)
	}
	return times
}
```

```go
func (e *EngineV2) Run(ctx context.Context) (*Stats, error) {
	var err error

	// Open source DB (Accede mode for old format compatibility)
	e.srcDB, err = mdbx.NewMDBX(log2.New()).
		Path(e.cfg.SourcePath + "/chaindata").
		MapSize(2 * datasize.TB).Accede().Open(ctx)
	if err != nil {
		return nil, fmt.Errorf("open source DB: %w", err)
	}
	defer e.srcDB.Close()

	// Open/create target DB with full table schema
	e.dstDB, err = mdbx.NewMDBX(log2.New()).
		Path(e.cfg.TargetPath + "/chaindata").
		WithTableCfg(func(_ kv.TableCfg) kv.TableCfg { return kv.ChaindataTablesCfg }).
		MapSize(4 * datasize.TB).Open(ctx)
	if err != nil {
		return nil, fmt.Errorf("open target DB: %w", err)
	}
	defer e.dstDB.Close()

	// Check for resume point
	if e.cfg.EnableJMT {
		if err := e.dstDB.View(ctx, func(tx kv.Tx) error {
			ver, _ := jmtstore.ReadJMTVersion(tx)
			if ver > 0 {
				e.newBlockNum = ver
				e.lastBlockNum = ver
				fmt.Printf("Resuming from block %d\n", ver)
			}
			return nil
		}); err != nil {
			return nil, err
		}
	}

	// Initialize JMT + LtHash
	if err := e.initStateCommitment(ctx); err != nil {
		return nil, err
	}

	// Detect source chain height
	if e.cfg.ToBlock == 0 {
		e.srcDB.View(ctx, func(tx kv.Tx) error {
			e.cfg.ToBlock = FindLatestBlock(tx)
			return nil
		})
	}
	e.stats.FromBlock = e.cfg.FromBlock
	e.stats.ToBlock = e.cfg.ToBlock

	// Initialize genesis if starting from 0
	if e.newBlockNum == 0 {
		if err := e.initGenesis(ctx); err != nil {
			return nil, fmt.Errorf("init genesis: %w", err)
		}
	}

	startFrom := e.lastBlockNum + 1
	if startFrom < 1 {
		startFrom = 1
	}

	// Progress ticker
	ticker := time.NewTicker(10 * time.Second)
	defer ticker.Stop()
	go func() {
		for range ticker.C {
			if e.cfg.ProgressFn != nil {
				e.cfg.ProgressFn(e.newBlockNum, e.cfg.ToBlock, e.stats.BlocksPerSecond())
			}
		}
	}()

	// Process in batches
	for batchStart := startFrom; batchStart <= e.cfg.ToBlock; batchStart += uint64(e.cfg.BatchSize) {
		select {
		case <-ctx.Done():
			return e.stats, ctx.Err()
		default:
		}
		batchEnd := batchStart + uint64(e.cfg.BatchSize) - 1
		if batchEnd > e.cfg.ToBlock {
			batchEnd = e.cfg.ToBlock
		}
		if err := e.processBatchV2(ctx, batchStart, batchEnd); err != nil {
			return e.stats, fmt.Errorf("batch %d-%d: %w", batchStart, batchEnd, err)
		}
	}

	return e.stats, nil
}
```

- [ ] **Step 3: Write `initStateCommitment()` — create JMT tree + LtHash digest**

```go
func (e *EngineV2) initStateCommitment(ctx context.Context) error {
	if !e.cfg.EnableJMT {
		return nil
	}

	// Read existing JMT root for resume
	var jmtRoot jmt.Hash
	e.dstDB.View(ctx, func(tx kv.Tx) error {
		jmtRoot, _ = jmtstore.ReadJMTRoot(tx)
		return nil
	})

	memStore := jmt.NewMemStore()
	var tree *jmt.Tree
	if jmtRoot == jmt.EmptyHash {
		tree = jmt.New(memStore)
	} else {
		tree = jmt.NewFromRoot(memStore, jmtRoot)
	}
	e.jmtCommit = commitment.NewJMTCommitment(tree)

	if e.cfg.EnableLtHash {
		var digest *lthash.Digest
		e.dstDB.View(ctx, func(tx kv.Tx) error {
			digest, _ = lthash.ReadLtHashDigest(tx, modules.LtHashDigest)
			return nil
		})
		if digest == nil {
			digest = lthash.New()
		}
		e.ltCommit = commitment.NewLtHashCommitment(digest)
	}
	return nil
}
```

```go
func (e *EngineV2) initGenesis(ctx context.Context) error {
	return e.dstDB.Update(ctx, func(tx kv.RwTx) error {
		// Create genesis state
		w := state.NewPlainStateWriterNoHistory(tx)
		r := state.NewPlainStateReader(tx)
		ibs := state.New(r)

		// Apply extra genesis state (hardfork allocs + system contracts)
		InitGenesisState(ibs)

		// Finalize genesis state
		rules := e.cfg.ChainConfig.Rules(0)
		if err := ibs.FinalizeTx(rules, w); err != nil {
			return fmt.Errorf("finalize genesis state: %w", err)
		}

		// Compute JMT + LtHash for genesis state
		var genesisRoot types.Hash
		var ltHashRoot types.Hash
		if e.jmtCommit != nil {
			// Collect dirty state and update JMT
			genesisRoot = e.computeStateRoot(ibs)
			if e.ltCommit != nil {
				ltHashRoot = e.ltCommit.Root()
			}
		}

		// Build genesis header
		header := &block.Header{
			Number:   uint256.NewInt(0),
			Time:     1678174066,
			Root:     genesisRoot,
			GasLimit: 30000000,
			BaseFee:  uint256.NewInt(params.InitialBaseFee),
		}
		header.LtHashRoot = ltHashRoot

		genesisBlock := block.NewBlock(header, nil)
		if err := rawdb.WriteBlock(tx, genesisBlock.(*block.Block)); err != nil {
			return err
		}
		if err := rawdb.WriteCanonicalHash(tx, genesisBlock.Hash(), 0); err != nil {
			return err
		}
		if err := rawdb.WriteTd(tx, genesisBlock.Hash(), 0, uint256.NewInt(0)); err != nil {
			return err
		}

		// Persist JMT state
		if e.jmtCommit != nil {
			e.flushJMT(tx, 0)
		}

		e.lastBlockTime = header.Time
		e.lastBlockNum = 0
		e.newBlockNum = 0
		return nil
	})
}
```

```go
func (e *EngineV2) processBatchV2(ctx context.Context, from, to uint64) error {
	return e.dstDB.Update(ctx, func(dstTx kv.RwTx) error {
		r := state.NewPlainStateReader(dstTx)
		return e.srcDB.View(ctx, func(srcTx kv.Tx) error {
			for srcNum := from; srcNum <= to; srcNum++ {
				blk, err := rawdb.ReadBlockByNumber(srcTx, srcNum)
				if err != nil || blk == nil {
					e.stats.BlocksMissing.Add(1)
					continue
				}
				srcHeader, ok := blk.Header().(*block.Header)
				if !ok {
					continue
				}

				// Fill gaps with empty blocks
				if e.cfg.FillGaps && e.lastBlockTime > 0 {
					gapTimes := CalcGapBlockTimes(e.lastBlockTime, srcHeader.Time, e.cfg.GapPeriod, e.cfg.GapTolerance)
					for _, gapTime := range gapTimes {
						e.newBlockNum++
						if err := e.writeEmptyBlock(dstTx, r, e.newBlockNum, gapTime); err != nil {
							return err
						}
					}
				}

				// Replay the actual block
				e.newBlockNum++
				if err := e.replayBlockV2(dstTx, r, blk, srcNum, e.newBlockNum); err != nil {
					e.stats.TxFailed.Add(1)
				}
				e.lastBlockTime = srcHeader.Time
				e.lastBlockNum = srcNum
			}
			// Persist JMT + LtHash at end of batch
			if e.jmtCommit != nil {
				e.flushJMT(dstTx, e.newBlockNum)
			}
			if e.ltCommit != nil {
				lthash.WriteLtHashDigest(dstTx, modules.LtHashDigest, e.ltCommit.Digest())
			}
			return nil
		})
	})
}

func (e *EngineV2) replayBlockV2(dstTx kv.RwTx, reader *state.PlainStateReader, blk *block.Block, srcNum, newNum uint64) error {
	e.stats.BlocksProcessed.Add(1)
	txs := blk.Transactions()
	srcHeader := blk.Header().(*block.Header)

	w := state.NewPlainStateWriterNoHistory(dstTx)
	ibs := state.New(reader)

	var replayedTxs []*transaction.Transaction
	blockNum := new(big.Int).SetUint64(srcNum)

	for _, tx := range txs {
		e.stats.TxTotal.Add(1)

		if to := tx.To(); to != nil && e.cfg.SkipAddresses[*to] {
			e.stats.TxSkipped.Add(1)
			continue
		}
		if tx.To() == nil && len(tx.Data()) == 0 {
			e.stats.TxSkipped.Add(1)
			continue
		}

		signer := transaction.MakeSigner(&params.ChainConfig{ChainID: big.NewInt(94)}, blockNum)
		from, err := transaction.Sender(signer, tx)
		if err != nil {
			e.stats.TxSkipped.Add(1)
			continue
		}

		value := tx.Value()
		if value != nil && !value.IsZero() {
			if ibs.GetBalance(from).Cmp(value) >= 0 {
				ibs.SubBalance(from, value)
				if tx.To() != nil {
					ibs.AddBalance(*tx.To(), value)
				}
			}
		}
		ibs.SetNonce(from, ibs.GetNonce(from)+1)

		if tx.To() == nil && len(tx.Data()) > 0 {
			contractAddr := crypto.CreateAddress(from, tx.Nonce())
			ibs.CreateAccount(contractAddr, true)
			ibs.SetCode(contractAddr, tx.Data())
		}

		replayedTxs = append(replayedTxs, tx)
		e.stats.TxReplayed.Add(1)
	}

	rules := e.cfg.ChainConfig.Rules(newNum)
	if err := ibs.FinalizeTx(rules, w); err != nil {
		return err
	}

	// Compute state root
	var stateRoot, ltRoot types.Hash
	if e.jmtCommit != nil {
		stateRoot = e.computeStateRoot(ibs)
	}
	if e.ltCommit != nil {
		ltRoot = e.ltCommit.Root()
	}

	// Build new header
	newHeader := &block.Header{
		ParentHash: e.lastParentHash(dstTx, newNum),
		Number:     uint256.NewInt(newNum),
		Time:       srcHeader.Time,
		Root:       stateRoot,
		GasLimit:   srcHeader.GasLimit,
		GasUsed:    srcHeader.GasUsed,
		Coinbase:   srcHeader.Coinbase,
		BaseFee:    srcHeader.BaseFee,
		Extra:      srcHeader.Extra,
		LtHashRoot: ltRoot,
	}

	newBlk := block.NewBlock(newHeader, replayedTxs)
	return e.writeNewBlock(dstTx, newBlk.(*block.Block), newNum)
}
```

```go
	// The dirty accounts from ibs are collected and fed to JMT
	return e.jmtCommit.Root()

func (e *EngineV2) flushJMT(tx kv.RwTx, blockNum uint64) {
	mdbxStore := jmtstore.NewMDBXStore(tx, jmtstore.JMTNodeTable)
	e.jmtCommit.Tree().FlushTo(mdbxStore)

	jmtstore.WriteJMTRoot(tx, jmtRoot)
	jmtstore.WriteJMTVersion(tx, blockNum)
	jmtstore.WriteJMTVersionRoot(tx, blockNum, jmtRoot)
	e.jmtCommit.Tree().SetVersion(blockNum)
}
```

```go
package era

import (
	"github.com/klauspost/compress/zstd"
	"sync"
)

// Zstd compression for EraE block data records.
// Version 2 EraE files use compressed records; version 1 is uncompressed.

const (
	VersionUncompressed = 1
	VersionZstd         = 2
	ZstdLevel           = 3 // zstd.SpeedDefault — good balance of speed vs ratio
)

var (
	encoderPool sync.Pool
	decoderPool sync.Pool
)

func getEncoder() *zstd.Encoder {
	if v := encoderPool.Get(); v != nil {
		return v.(*zstd.Encoder)
	}
	enc, _ := zstd.NewWriter(nil, zstd.WithEncoderLevel(zstd.EncoderLevel(ZstdLevel)))
	return enc
}

func putEncoder(enc *zstd.Encoder) { encoderPool.Put(enc) }

func getDecoder() *zstd.Decoder {
	if v := decoderPool.Get(); v != nil {
		return v.(*zstd.Decoder)
	}
	dec, _ := zstd.NewReader(nil)
	return dec
}

func putDecoder(dec *zstd.Decoder) { decoderPool.Put(dec) }

// CompressBlock compresses block data using Zstd.
func CompressBlock(data []byte) []byte {
	enc := getEncoder()
	defer putEncoder(enc)
	return enc.EncodeAll(data, make([]byte, 0, len(data)/2))
}

// DecompressBlock decompresses Zstd-compressed block data.
func DecompressBlock(compressed []byte) ([]byte, error) {
	dec := getDecoder()
	defer putDecoder(dec)
	return dec.DecodeAll(compressed, make([]byte, 0, len(compressed)*3))
}
```

```go
type CheckpointEntry struct {

	Hash   string `json:"hash"`

// RunPostExport creates snapshot, EraE segments, and checkpoint after replay.
func (e *EngineV2) RunPostExport(ctx context.Context) error {
	if e.cfg.SnapshotAtEnd {
		fmt.Println("Creating final snapshot...")
		if err := e.createSnapshot(ctx); err != nil {
			return fmt.Errorf("snapshot: %w", err)
		}
	}

	if e.cfg.ExportEraE {
		fmt.Println("Exporting EraE segments...")
		if err := e.exportEraE(ctx); err != nil {
			return fmt.Errorf("era export: %w", err)
		}
	}

	// Always write checkpoint
	return e.writeCheckpoint(ctx)
}

func (e *EngineV2) createSnapshot(ctx context.Context) error {
	return e.dstDB.Update(ctx, func(tx kv.RwTx) error {
		meta := &snapshot.Meta{
			BlockNumber: e.newBlockNum,
		}
		// Count tables
		e.dstDB.View(ctx, func(roTx kv.Tx) error {
			meta.AccountCount, _ = countTableEntries(roTx, "Account")
			meta.StorageCount, _ = countTableEntries(roTx, "Storage")
			meta.CodeCount, _ = countTableEntries(roTx, "Code")
			return nil
		})
		meta.CreatedAt = e.stats.StartTime.Unix()
		return snapshot.SaveSnapshot(tx, meta)
	})
}

func (e *EngineV2) exportEraE(ctx context.Context) error {
	eraDir := filepath.Join(e.cfg.TargetPath, "era")
	os.MkdirAll(eraDir, 0700)
	exporter := torrentsync.NewExporter(e.dstDB, eraDir, e.cfg.EraESegmentSize)
	if _, err := exporter.ExportRange(ctx, 0, e.newBlockNum); err != nil {
		return err
	}
	// Generate manifest
	svc := torrentsync.NewService(&conf.TorrentSyncCfg{
		SegmentSize: e.cfg.EraESegmentSize,
	}, e.dstDB, eraDir)
	return svc.GenerateManifest()
}

func (e *EngineV2) writeCheckpoint(ctx context.Context) error {
	var hash string
	e.dstDB.View(ctx, func(tx kv.Tx) error {
		h, _ := rawdb.ReadCanonicalHash(tx, e.newBlockNum)
		hash = fmt.Sprintf("0x%x", h)
		return nil
	})
	cp := CheckpointEntry{Number: e.newBlockNum, Hash: hash}
	data, _ := json.MarshalIndent(cp, "", "  ")
	path := filepath.Join(e.cfg.TargetPath, "checkpoint.json")
	return os.WriteFile(path, data, 0644)
}

func countTableEntries(tx kv.Tx, table string) (uint64, error) {
	cursor, err := tx.Cursor(table)
	if err != nil {
		return 0, err
	}
	defer cursor.Close()
	count, err := cursor.Count()
	return count, err
}
```
